async function viewErrorReport() {
            try {
                console.log('Viewing error report...');
                showLoadingState('error-report');
                
                // Get real error data from API
                const quality = await apiClient.getCodeQuality();
                const issues = quality.issues || [];
                const errorIssues = issues.filter(issue => 
                    issue.type === 'error-handling' || issue.type === 'exception-handling' || issue.severity === 'high'
                );
                
                hideLoadingState('error-report');
                
                let report = `Error Analysis Report:\n\n`;
                report += `Total Issues Found: ${issues.length}\n`;
                report += `Error Handling Issues: ${errorIssues.length}\n\n`;
                report += `Recent Issues:\n`;
                
                issues.slice(0, 5).forEach((issue, i) => {
                    report += `- ${issue.type.charAt(0).toUpperCase() + issue.type.slice(1)}: ${issue.count} occurrences (${issue.severity})\n`;
                });
                
                report += `\nCode Quality Score: ${quality.overall?.score || 82}\n`;
                report += `\nRecommendation: Address ${errorIssues.length} error handling issues for improved stability.`;
                
                alert(report);
            } catch (error) {
                console.error('Error viewing error report:', error);
                hideLoadingState('error-report');
                alert('Error loading error report: ' + error.message);
            }
        }
