# Simplebeacon — Zorin Linux Quick Reference (Thumb Drive Edition)
# Full guide: ZORIN-LINUX-SETUP.md

# ═══════ THUMB DRIVE PATHS (Zorin auto-mounts under /media) ═══════
# IMPORTANT: Copy the ENTIRE "CascadeProjects" folder to your thumb drive.
# Do NOT copy only "ai-platform/" — the CLI scanner lives outside it.
#
# Find your drive name after plugging it in:
#   ls /media/$USER/
# Example:
#   cd /media/$USER/SANDISK/CascadeProjects

# ═══════ SETUP (run once from the thumb drive) ═══════
cd /media/$USER/YOUR_DRIVE_NAME/CascadeProjects
chmod +x setup-zorin.sh
./setup-zorin.sh                     # auto-detects thumb drive path, installs Node.js, links CLI

# ═══════ VERIFY ═══════
simplebeacon --help                  # confirm CLI is linked to /usr/local/bin
simplebeacon-mcp --smoke-test        # confirm MCP server works
simplebeacon scan --gate --offline   # test scan on this repo

# ═══════ DAILY USE ═══════
# Scan current directory (any project on Zorin)
simplebeacon scan --gate --offline

# Scan a specific project
simplebeacon scan --path /home/user/my-project --gate --offline

# Full coverage (every file, not just src/)
simplebeacon scan --path /home/user/my-project --full --gate --offline

# JSON report for CI
simplebeacon scan --gate --format json --output .simplebeacon/report.json

# Check last gate result
simplebeacon gate status

# Start MCP server (Cursor / Claude Desktop)
simplebeacon-mcp --offline

# ═══════ SCAN A WEBSITE ═══════
# Option A: CLI (terminal)
cd /media/$USER/YOUR_DRIVE_NAME/CascadeProjects
./scan-website.sh https://your-domain.com --gate

# Option B: Dashboard (browser)
# Open the dashboard, go to Analyze, click "Website URL" tab,
# enter https://your-domain.com, then click Run analysis.

# With JSON report output:
./scan-website.sh https://your-domain.com --gate --output report.json

# Keep downloaded files for manual inspection:
./scan-website.sh https://your-domain.com --keep

# ═══════ DASHBOARD (optional, runs from thumb drive) ═══════
# One command — starts server + waits for ready + opens browser:
cd /media/$USER/YOUR_DRIVE_NAME/CascadeProjects && ./start-dashboard.sh

# Manual start (if you prefer):
cd /media/$USER/YOUR_DRIVE_NAME/CascadeProjects/ai-platform && npm start
# Then open: http://localhost:54355/simplebeacon-dashboard/

# ═══════ TROUBLESHOOTING ═══════
# "command not found"     → re-run ./setup-zorin.sh or check /usr/local/bin
# Permission denied       → chmod +x setup-zorin.sh
#                         → If still denied (FAT32 thumb drive), use: bash setup-zorin.sh
# Can't find drive path   → run: ls /media/$USER/
# False positives         → edit .simplebeacon/config.json, add paths to exclude
# Scripts missing          → Make sure you copied the WHOLE repo, not just ai-platform/
